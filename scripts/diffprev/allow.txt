# diff-prev allowlist (scripts/diffprev, make diff-prev): intentional output
# changes of the current compiler against the previous release.
#
# One entry per line: <case name> # <reason>
# The case name is the one diff-prev prints (also with -v). Remove entries
# when the seed moves past the change.
#
# Rewording rules (scripts/diagallow): E#### "old text" => "new text" # reason
# allowlist every DIAG-CHANGED case that differs only in diagnostics at the
# same position with the same code, reworded as a rule says (the texts are
# Go string literals: old is a substring of the previous message, new of the
# current one). `driver "old" => "new" # reason` rewords the compiler's own
# non-diagnostic lines: old replaced by new must give the current line.

E1001 "expected type, found ','" => "a var declaration declares one name; write separate var declarations" # v0.3, decision 157: `var a, b T`; same position (the ',')
E3002 ", which outlives it" => ": each parameter is its own region" # v0.3, decision 157: a store between parameters at the same depth (decision 44) states the rule instead of "(depth N) ... (depth N), which outlives it"
